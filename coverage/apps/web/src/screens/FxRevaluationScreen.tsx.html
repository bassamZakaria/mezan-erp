import { Fragment, useEffect, useState } from "react";
import {
  Button,
  Card,
  Checkbox,
  Chip,
  DocumentList,
  EmptyState,
  StatCard,
  TextInput,
  VarianceBeam,
  formatNumber,
  type DocumentListColumn,
} from "@mezan/ui";
import type { CurrencyBalanceGroupDTO, FxRevaluationDetailDTO } from "@mezan/contracts";
import { foreignCurrencies, nextDay } from "@mezan/domain";
import {
  useAllowedCurrencies,
  useFxRevaluationPost,
  useFxRevaluationPreview,
  usePartyBalancesByCurrency,
} from "../data/hooks";
import { useAppTranslation } from "../i18n";
import { monthEndIso } from "../lib/dates";
import { useUrlNumber } from "../lib/url-state";
 
function fmt(value: string, digits = 2): string {
  return formatNumber(Number(value), { minimumFractionDigits: digits, maximumFractionDigits: digits });
}
 
/**
 * إعادة تقييم العملات الأجنبية — FX Revaluation (docs/archive/catalog/04). Revalues open
 * foreign-currency balances at the approved closing rate and posts the FX
 * gain/loss — auto-reversing the next day by default. Runs the REAL
 * `computeFxRevaluation` / `postFxRevaluation` domain services server-side
 * via `POST /revaluation/preview` and `/revaluation/post`.
 */
const PAGE_SIZES = [10, 25, 50, 100] as const;
 
export function FxRevaluationScreen() {
  const { t } = useAppTranslation();
  const previewMutation = useFxRevaluationPreview();
  const postMutation = useFxRevaluationPost();
  const allowedCurrencies = useAllowedCurrencies();
  const balancesQuery = usePartyBalancesByCurrency();
  const baseCurrency = allowedCurrencies.data?.base ?? "EGP";
  const currencyCodes = foreignCurrencies(baseCurrency).map((c) => c.code);
  const balances = balancesQuery.data ?? null;
  const [toDate, setToDate] = useState(monthEndIso());
  const [autoReverse, setAutoReverse] = useState(true);
  const [selected, setSelected] = useState<string[]>([]);
  const [error, setError] = useState<string | null>(null);
  const [posted, setPosted] = useState<{ entryNo: string | null; reversalNo: string | null } | null>(null);
  const [page, setPage] = useUrlNumber("page", 1);
  const [pageSize, setPageSize] = useUrlNumber("size", 25);
 
  function toggleCurrency(c: string) {
    setSelected((prev) => (prev.includes(c) ? prev.filter((x) => x !== c) : [...prev, c]));
  }
 
  async function runPreview() {
    setError(null);
    setPosted(null);
    try {
      await previewMutation.mutateAsync({ date: toDate, currencies: selected, autoReverse });
    } catch (err) {
      setError(err instanceof Error ? err.message : t("fx.theRevaluationCouldNotBe"));
    }
  }
 
  async function runPost() {
    setError(null);
    try {
      const result = await postMutation.mutateAsync({ date: toDate, currencies: selected, autoReverse });
      setPosted({ entryNo: result.entryNo, reversalNo: result.reversalNo });
      await runPreview();
    } catch (err) {
      setError(err instanceof Error ? err.message : t("fx.theRevaluationCouldNotBe2"));
    }
  }
 
  const columns: DocumentListColumn<FxRevaluationDetailDTO>[] = [
    { key: "account", header: t("fx.account"), render: (d) => <span className="font-semibold text-ink">{d.label ?? d.accountId}</span> },
    { key: "currency", header: t("fx.currency"), render: (d) => <span className="mz-num text-ink-muted">{d.currency}</span> },
    { key: "foreign", header: t("fx.foreignBalance"), render: (d) => <span className="mz-num">{fmt(d.foreignAmount)}</span> },
    { key: "rate", header: t("fx.closingRate"), render: (d) => <span className="mz-num">{fmt(d.closingRate, 4)}</span> },
    { key: "booked", header: t("fx.carryingValue"), render: (d) => <span className="mz-num">{fmt(d.bookedBase)}</span> },
    { key: "revalued", header: t("fx.afterRevaluation"), render: (d) => <span className="mz-num">{fmt(d.revaluedBase)}</span> },
    {
      key: "gl",
      header: t("fx.differenceGainLoss"),
      align: "end",
      render: (d) => (
        <VarianceBeam value={Number(d.gainLoss)} className="font-bold">
          <span className="mz-num">{fmt(d.gainLoss)}</span>
        </VarianceBeam>
      ),
    },
  ];
 
  const p = previewMutation.data ?? null;
  const details = p?.details ?? [];
  const pageCount = Math.max(1, Math.ceil(details.length / pageSize));
  const pageRows = details.slice((page - 1) * pageSize, page * pageSize);
  useEffect(() => {
    if (page > pageCount) setPage(1);
  }, [page, pageCount, setPage]);
 
  return (
    <div className="w-full">
      <div className="mb-[16px]">
        <h1 className="mt-[5px] font-display text-2xl font-bold text-ink">{t("fx.foreignCurrencyRevaluation")}</h1>
        <p className="mt-[5px] max-w-[70ch] text-[12.5px] text-ink-muted">{t("fx.thisPostsAnEntryRestating")}</p>
      </div>
 
      <div className="mb-[18px] rounded-lg border border-border bg-surface p-[16px_18px]">
        <div className="flex flex-wrap items-end gap-[16px]">
          <label className="flex flex-col gap-[4px] text-[12px] text-ink-muted">{t("fx.asOfEntryDate")}
            <TextInput size="sm" className="w-[160px]" type="date" aria-label={t("fx.asOf")} value={toDate} onChange={(e) => setToDate(e.target.value)} />
          </label>
          <Checkbox
            className="items-center"
            checked={autoReverse}
            onChange={(e) => setAutoReverse(e.target.checked)}
            label={`${t("fx.reversesAutomaticallyReversingEntryOn")} ${nextDay(toDate)})`}
          />
          <div className="flex-1" />
          <Button variant="primary" onClick={runPreview}>{t("fx.preview")}</Button>
          <Button onClick={runPost}>{t("fx.postRevaluation")}</Button>
        </div>
        <div className="mt-[12px] flex flex-wrap items-center gap-[8px]">
          <span className="text-[12px] text-ink-muted">{t("fx.currencies")}</span>
          <Chip toggle active={selected.length === 0} onClick={() => setSelected([])}>{t("common.all")}</Chip>
          {currencyCodes.map((c) => (
            <Chip key={c} toggle active={selected.includes(c)} onClick={() => toggleCurrency(c)}>{c}</Chip>
          ))}
        </div>
      </div>
 
      {error ? <div role="alert" className="mb-[14px] rounded-md border border-danger bg-danger-soft px-[12px] py-[10px] text-[12.5px] text-danger-ink">{error}</div> : null}
      {posted ? (
        <div role="status" className="mb-[14px] rounded-md border border-success bg-success-soft px-[12px] py-[10px] text-[12.5px] text-success-ink">
          {t("fx.theRevaluationEntryHasBeen")} {posted.entryNo ?? ""}{posted.reversalNo ? ` ${t("fx.withReversal", { number: posted.reversalNo })}` : ""}.
        </div>
      ) : null}
 
      {p ? (
        <>
          <div className="mb-[16px] grid grid-cols-1 gap-[14px] sm:grid-cols-3">
            <StatCard label={t("fx.totalGains")} value={<span className="mz-num">{fmt(p.totalGain)}</span>} />
            <StatCard label={t("fx.totalLosses")} value={<span className="mz-num">{fmt(p.totalLoss)}</span>} />
            <StatCard label={t("fx.reversalDate")} value={<span className="mz-num">{p.autoReverse ? p.reversalDate : "—"}</span>} />
          </div>
          <DocumentList<FxRevaluationDetailDTO>
            title=""
            columns={columns}
            rows={pageRows}
            loading={balancesQuery.isLoading}
            rowKey={(d) => `${d.accountId}-${d.currency}`}
            page={page}
            pageCount={pageCount}
            onPageChange={setPage}
            pageSize={pageSize}
            pageSizeOptions={PAGE_SIZES}
            onPageSizeChange={(size) => {
              setPageSize(size);
              setPage(1);
            }}
            pageSummary={<>{t("common.show")} <span className="mz-num">{pageRows.length}</span> {t("fx.balancesToRevalue")}</>}
            emptyState={<EmptyState title={t("fx.noOpenForeignCurrencyBalances")} description={t("fx.emptyBody")} />}
          />
        </>
      ) : (
        <div className="rounded-lg border border-dashed border-border p-8 text-center text-sm text-ink-muted">{t("fx.pressPreviewToListThe")}</div>
      )}
 
      {/*
        R1.16 — what the revaluation above is actually exposed to. A balance
        collapsed to base loses which currency it is denominated in, so this
        reports both: the base figure the control account holds, and the
        figure the customer or supplier owes in the currency they were billed.
      */}
      <section className="mt-[24px]">
        <h2 className="font-display text-lg font-bold text-ink">{t("fx.balances.title")}</h2>
        <p className="mt-[4px] max-w-[70ch] text-[12.5px] text-ink-muted">{t("fx.balances.description")}</p>
 
        {balancesQuery.isError ? (
          <div role="alert" className="mt-[12px] rounded-md border border-danger bg-danger-soft px-[12px] py-[10px] text-[12.5px] text-danger-ink">
            {t("fx.balances.couldNotLoad")}
          </div>
        ) : null}
 
        <div className="mt-[14px] grid grid-cols-1 gap-[16px] lg:grid-cols-2">
          <CurrencyBalanceTable
            heading={t("fx.balances.receivables")}
            groups={balances?.receivables ?? []}
            totalBase={balances?.receivableTotalBase ?? "0"}
          />
          <CurrencyBalanceTable
            heading={t("fx.balances.payables")}
            groups={balances?.payables ?? []}
            totalBase={balances?.payableTotalBase ?? "0"}
          />
        </div>
      </section>
    </div>
  );
}
 
function CurrencyBalanceTable({
  heading,
  groups,
  totalBase,
}: {
  heading: string;
  groups: readonly CurrencyBalanceGroupDTO[];
  totalBase: string;
}) {
  const { t } = useAppTranslation();
  const headCell =
    "whitespace-nowrap border-b border-border bg-surface-2 px-[13px] py-[9px] text-start text-[11px] font-bold uppercase tracking-[0.2px] text-ink-muted";
 
  return (
    <Card className="overflow-hidden">
      <div className="flex flex-wrap items-baseline justify-between gap-[8px] border-b border-border px-[14px] py-[10px]">
        <h3 className="text-[13px] font-bold text-ink">{heading}</h3>
        <span className="text-[11.5px] text-ink-muted">
          {t("fx.balances.totalBase")} <span className="mz-num font-bold text-ink">{fmt(totalBase)}</span>
        </span>
      </div>
      <div className="overflow-x-auto">
        <table className="w-full min-w-[420px] border-collapse text-[13px]">
          <thead>
            <tr>
              <th scope="col" className={headCell}>{t("fx.balances.party")}</th>
              <th scope="col" className={headCell}>{t("fx.currency")}</th>
              <th scope="col" className={headCell}>{t("fx.balances.transactionAmount")}</th>
              <th scope="col" className={headCell}>{t("fx.balances.baseAmount")}</th>
              <th scope="col" className={headCell}>{t("fx.balances.documents")}</th>
            </tr>
          </thead>
          <tbody>
            {groups.map((group) => (
              <Fragment key={group.currency}>
                <tr className="border-b border-border bg-surface-2">
                  <td className="px-[13px] py-[8px] font-bold text-ink" colSpan={2}>{group.currency}</td>
                  <td className="mz-num px-[13px] py-[8px] font-bold text-ink">{fmt(group.outstandingTransaction)}</td>
                  <td className="mz-num px-[13px] py-[8px] font-bold text-ink">{fmt(group.outstandingBase)}</td>
                  <td className="px-[13px] py-[8px]" />
                </tr>
                {group.parties.map((party) => (
                  <tr key={`${group.currency}-${party.partyId}`} className="border-b border-border last:border-b-0">
                    <td className="px-[13px] py-[8px] text-ink">{party.partyName}</td>
                    <td className="mz-num px-[13px] py-[8px] text-ink-muted">{party.currency}</td>
                    <td className="mz-num px-[13px] py-[8px] text-ink">{fmt(party.outstandingTransaction)}</td>
                    <td className="mz-num px-[13px] py-[8px] text-ink-muted">{fmt(party.outstandingBase)}</td>
                    <td className="mz-num px-[13px] py-[8px] text-ink-muted">{party.documentCount}</td>
                  </tr>
                ))}
              </Fragment>
            ))}
          </tbody>
        </table>
        {groups.length === 0 ? (
          <div className="p-6 text-center text-[12.5px] text-ink-muted">{t("fx.balances.empty")}</div>
        ) : null}
      </div>
    </Card>
  );
}
